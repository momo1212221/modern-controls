cmake_minimum_required(VERSION 3.20)
project(modern_controls LANGUAGES C CXX)

if(NOT OpenRACMod_DIR)
  if(DEFINED OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
    string(JSON sdk_version GET "${lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
    if(NOT EXISTS "${OpenRACMod_DIR}/OpenRACModConfig.cmake")
      message(FATAL_ERROR "No OpenRAC mod SDK ${sdk_version}: run python3 tools/fetch_sdk.py first")
    endif()
  endif()
endif()
find_package(OpenRACMod 1.0 REQUIRED)
include(CTest)

openrac_add_mod(modern_controls MANIFEST openrac-mod.json SOURCES src/modern_controls.cpp)
target_include_directories(modern_controls PRIVATE src)
target_link_libraries(modern_controls PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(modern_controls PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(modern_controls PRIVATE /utf-8)
else()
  target_compile_options(modern_controls PRIVATE -Wall -Wextra)
endif()

if(BUILD_TESTING)
  add_executable(modern_controls_test tests/controls_test.cpp)
  target_include_directories(modern_controls_test PRIVATE src)
  target_link_libraries(modern_controls_test PRIVATE OpenRAC::mod)
  set_target_properties(modern_controls_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_test(NAME modern_controls COMMAND modern_controls_test)
endif()

if(BUILD_TESTING AND UNIX)
  add_executable(modern_controls_load_test tests/load_test.cpp)
  target_link_libraries(modern_controls_load_test PRIVATE OpenRAC::mod ${CMAKE_DL_LIBS})
  set_target_properties(modern_controls_load_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_dependencies(modern_controls_load_test modern_controls)
  add_test(NAME modern_controls_load COMMAND modern_controls_load_test $<TARGET_FILE:modern_controls>)
endif()

get_target_property(controls_package modern_controls OPENRAC_MOD_PACKAGE_DIR)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${controls_package}/LICENSE" COPYONLY)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/README.md" "${controls_package}/README.md" COPYONLY)
